### 数据库模型 `models.py`（新建表 user_profile）

> 
> 一期：全局单条记录，代表当前使用者；二期扩展增加 user_id 做多用户。

```
# app/modules/personal_center/models.py
from sqlalchemy import Column, Integer, String, Text
from app.core.db import Base

class UserProfile(Base):
    __tablename__ = "user_profile"

    id = Column(Integer, primary_key=True, autoincrement=True)
    avatar_url = Column(String(255), nullable=True, comment="头像访问地址")
    signature = Column(Text, nullable=True, comment="个性签名")
    phone = Column(String(20), nullable=True, comment="手机号")
    username = Column(String(50), nullable=True, comment="账号名")
    password_hash = Column(String(255), nullable=False, comment="密码哈希，不存明文")
    create_time = Column(String, nullable=True)
    update_time = Column(String, nullable=True)
```

from pydantic import BaseModel, Field
from typing import Optional

# 获取个人资料返回
class ProfileResp(BaseModel):
    avatar_url: Optional[str]
    signature: Optional[str]
    phone: Optional[str]
    username: Optional[str]

# 更新基础资料
class ProfileUpdateReq(BaseModel):
    signature: Optional[str] = Field(None, max_length=100, description="个性签名，最多100字符")
    phone: Optional[str] = Field(None, description="手机号")

# 修改密码
class PasswordChangeReq(BaseModel):
    old_password: str
    new_password: str

### repository.py

```
# app/modules/personal_center/repository.py
from sqlalchemy import select
from app.core.db import db_session
from app.modules.personal_center.models import UserProfile

class ProfileRepo:
    async def get_profile(self):
        """一期：只取第一条（模拟当前登录用户）"""
        async with db_session() as session:
            res = await session.execute(select(UserProfile).limit(1))
            return res.scalar_one_or_none()

    async def update_profile(self, profile_obj):
        async with db_session() as session:
            session.add(profile_obj)
            await session.commit()
            await session.refresh(profile_obj)
            return profile_obj

profile_repo = ProfileRepo()
```

### service.py

业务：头像存储、密码哈希处理、资料更新

> 
> 头像上传：后端保存到`resources/upload/avatar`，返回可访问 url；密码**禁止存储明文，使用 bcrypt 哈希**

```
# app/modules/personal_center/service.py
import bcrypt
import os
from datetime import datetime
from fastapi import UploadFile
from app.modules.personal_center.repository import profile_repo
from app.modules.personal_center.models import UserProfile
from app.config import settings

class ProfileService:
    async def get_my_profile(self):
        profile = await profile_repo.get_profile()
        if not profile:
            # 初始化一条默认记录
            new_profile = UserProfile(
                username="默认用户",
                password_hash=self._hash_password("123456"),
                create_time=datetime.now().strftime("%Y‑%m‑%d %H:%M:%S")
            )
            return await profile_repo.update_profile(new_profile)
        return profile

    def _hash_password(self, raw_pwd: str) -> str:
        salt = bcrypt.gensalt()
        return bcrypt.hashpw(raw_pwd.encode("utf‑8"), salt).decode("utf‑8")

    def _verify_password(self, raw_pwd: str, hash_pwd: str) -> bool:
        return bcrypt.checkpw(raw_pwd.encode("utf‑8"), hash_pwd.encode("utf‑8"))

    async def update_basic_info(self, signature: str | None, phone: str | None):
        profile = await self.get_my_profile()
        if signature is not None:
            profile.signature = signature
        if phone is not None:
            profile.phone = phone
        profile.update_time = datetime.now().strftime("%Y‑%m‑%d %H:%M:%S")
        return await profile_repo.update_profile(profile)

    async def upload_avatar(self, file: UploadFile):
        """保存头像文件，返回访问url"""
        save_dir = os.path.join(settings.RESOURCES_PATH, "upload/avatar")
        os.makedirs(save_dir, exist_ok=True)
        filename = f"{datetime.now().timestamp()}_{file.filename}"
        save_path = os.path.join(save_dir, filename)
        content = await file.read()
        with open(save_path, "wb") as f:
            f.write(content)
        avatar_url = f"/resources/upload/avatar/{filename}"
        profile = await self.get_my_profile()
        profile.avatar_url = avatar_url
        await profile_repo.update_profile(profile)
        return avatar_url

    async def change_password(self, old_password: str, new_password: str):
        profile = await self.get_my_profile()
        if not self._verify_password(old_password, profile.password_hash):
            raise Exception("原密码错误")
        profile.password_hash = self._hash_password(new_password)
        profile.update_time = datetime.now().strftime("%Y‑%m‑%d %H:%M:%S")
        await profile_repo.update_profile(profile)

profile_service = ProfileService()
```

### router.py 接口

```
# app/modules/personal_center/router.py
from fastapi import APIRouter, UploadFile, File
from app.modules.personal_center.schemas import ProfileResp, ProfileUpdateReq, PasswordChangeReq
from app.modules.personal_center.service import profile_service

router = APIRouter(prefix="/api/profile", tags=["个人中心"])

@router.get("/info", response_model=ProfileResp)
async def get_profile():
    """获取个人资料"""
    return await profile_service.get_my_profile()

@router.put("/info", response_model=ProfileResp)
async def update_profile(req: ProfileUpdateReq):
    """更新签名、手机号"""
    return await profile_service.update_basic_info(signature=req.signature, phone=req.phone)

@router.post("/avatar")
async def upload_avatar(file: UploadFile = File(...)):
    """上传头像"""
    url = await profile_service.upload_avatar(file)
    return {"avatar_url": url}

@router.post("/change‑pwd")
async def change_password(req: PasswordChangeReq):
    """修改密码"""
    await profile_service.change_password(req.old_password, req.new_password)
    return {"msg":"密码修改成功"}
```

## 3、一期约束与二期规划

### ✅一期（当前实现，无登录）

1. 数据库只存 1 条用户配置；没有多用户、没有登录、没有 token；
2. 右上角头像下拉，弹窗编辑资料、上传头像、修改密码；
3. 密码做 bcrypt 哈希，不存储明文；头像存本地资源目录；
4. 全部接口直接访问，不做鉴权。

### 📌二期扩展（后续增加登录模块）

1. 新增用户表、注册、登录、JWT token 鉴权；
2. `user_profile`表增加外键`user_id`支持多用户；
3. 接口增加鉴权依赖`common/deps.py`，每个接口校验 token。
